    <!-- JavaScript Logic -->
    <script>
        let currentRates = {};

        window.addEventListener('DOMContentLoaded', () => {
            fetchRates();
        });

        async function fetchRates() {
            const from = document.getElementById('fromCurrency').value;
            try {
                const res = await fetch(`https://api.frankfurter.app/latest?from=${from}`);
                const data = await res.json();
                currentRates = data.rates;
                currentRates[from] = 1; // base rate equivalence
                convertCurrency();
            } catch (err) {
                console.error(err);
                document.getElementById('exchangeRateInfo').innerText = "Failed to load live rates. Using offline estimates.";
            }
        }

        async function convertCurrency() {
            const from = document.getElementById('fromCurrency').value;
            const to = document.getElementById('toCurrency').value;
            const amount = parseFloat(document.getElementById('amountInput').value) || 0;

            // If base currency changed, re-fetch rates for accuracy
            if (!currentRates[to] && from) {
                await fetchRates();
                return;
            }

            let rate = 1;
            if (from === to) {
                rate = 1;
            } else if (currentRates[to]) {
                rate = currentRates[to];
            } else {
                // Fallback direct calculation lookup if needed
                rate = 18.5; // fallback approximation if API lags
            }

            const converted = amount * rate;
            document.getElementById('resultValue').innerText = `${converted.toFixed(2)} ${to}`;
            document.getElementById('exchangeRateInfo').innerText = `1 ${from} = ${rate.toFixed(4)} ${to} (Live Market Rate)`;
            
            calculateExtras(converted);
        }

        function calculateExtras(baseAmount) {
            if (baseAmount === undefined) {
                const amount = parseFloat(document.getElementById('amountInput').value) || 0;
                const to = document.getElementById('toCurrency').value;
                const rate = currentRates[to] || 1;
                baseAmount = amount * rate;
            }

            const discount = parseFloat(document.getElementById('discountInput').value) || 0;
            const tax = parseFloat(document.getElementById('taxInput').value) || 0;
            const to = document.getElementById('toCurrency').value;

            let adjusted = baseAmount - (baseAmount * (discount / 100));
            adjusted = adjusted + (adjusted * (tax / 100));

            document.getElementById('finalPriceValue').innerText = `${adjusted.toFixed(2)} ${to}`;
        }
    </script>
